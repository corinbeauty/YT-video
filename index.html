/** * 門店內訓影音自動巡檢系統 - 雲端接收端完整版 */ function doPost(e) { try { // 1. 解析來自 HTML 網頁的資料 var data = JSON.parse(e.postData.contents); var storeName = data.storeName || "未知門店"; var slot = data.slot || "未知時段"; var timestamp = data.timestamp || ""; var videoTitle = data.videoTitle || "無標題"; var phaseTag = data.phaseTag || "巡檢"; var imageBase64 = data.imageBase64; // 2. 移除 Base64 header，並轉換為 Blob 圖片物件 var base64Data = imageBase64.replace(/^data:image\/(png|jpeg);base64,/, ""); var blob = Utilities.newBlob(Utilities.base64Decode(base64Data), "image/jpeg"); // 3. 建立檔案名稱 (格式: 2026-08-01-12-00-00_門店名稱_時段_階段.jpg) var safeTimestamp = timestamp.replace(/[: ]/g, "-"); var fileName = safeTimestamp + "_" + storeName + "_" + slot + "_" + phaseTag + ".jpg"; blob.setName(fileName); // 4. 綁定您指定的資料夾 ID var targetFolderId = "1KeOX9WilM7VhhAadhCYEse7g0Xthtskl"; var targetFolder = DriveApp.getFolderById(targetFolderId); // 5. 將圖片檔案儲存到該資料夾內 var file = targetFolder.createFile(blob); // 6. 回傳成功訊息給網頁 return ContentService.createTextOutput(JSON.stringify({ status: "success", fileUrl: file.getUrl() })).setMimeType(ContentService.MimeType.JSON); } catch (error) { // 錯誤日誌生成器：如果發生異常，在雲端硬碟根目錄寫入錯誤檔案 try { DriveApp.createFile("Error_Log_" + new Date().getTime() + ".txt", error.toString()); } catch(e) {} // 回傳錯誤訊息給網頁前端 return ContentService.createTextOutput(JSON.stringify({ status: "error", message: error.toString() })).setMimeType(ContentService.MimeType.JSON); } } /** * ！！！非常重要！！！ * 請在 GAS 編輯器上方工具列選擇此「testAuth」函式並點擊「執行(Run)」 * 藉此觸發 Google 雲端硬碟的權限審查畫面 */ function testAuth() { var folderId = "1KeOX9WilM7VhhAadhCYEse7g0Xthtskl"; var folder = DriveApp.getFolderById(folderId); Logger.log("恭喜！授權已成功開啟！您指定的資料夾名稱為：" + folder.getName()); }
